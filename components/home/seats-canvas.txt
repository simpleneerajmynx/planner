"use client"

/**
 * SeatsCanvas — ultra-performance seat renderer
 *
 * Instead of 1 React component per seat (→ 1000+ Konva nodes, slow),
 * we use a SINGLE Konva <Shape> that draws every seat in one
 * requestAnimationFrame via raw Canvas 2D API.
 *
 * Hit detection uses a spatial hash grid (cell = 2×SEAT_STEP px).
 * Mouse events on the Stage are forwarded here; we find the nearest
 * seat in O(1) and handle hover / click ourselves.
 *
 * Result: smooth 60 fps even with 2000+ seats.
 */

import React, { useEffect, useRef, useCallback, useMemo } from "react"
import { Layer, Shape } from "react-konva"
import Konva from "konva"
import { Seat, SEAT_COLORS } from "@/types/seat"
import { useBookingStore } from "@/store/bookingStore"
import { SEAT_STEP } from "@/data/stadiumData"

// ─── constants ───────────────────────────────────────────────────────────────
const R = 4 // world-space radius (normal)
const R_SEL = 4.8 // selected
const R_HOV = 5.2 // hovered
const CELL = SEAT_STEP * 2 // spatial grid cell size

// Colours (pre-parsed for speed)
const C_AVAIL = SEAT_COLORS.available // '#1d4ed8'
const C_SEL = SEAT_COLORS.selected // '#16a34a'
const C_TAKEN = SEAT_COLORS.taken // '#2d2d3a'
const C_VIP = SEAT_COLORS.vip // '#b45309'
const C_HOV = SEAT_COLORS.hover // '#60a5fa'
const C_SEL_GL = "#22c55e" // glow colour for selected

// ─── spatial hash ─────────────────────────────────────────────────────────────
type Grid = Map<string, Seat[]>

function buildGrid(seats: Seat[]): Grid {
  const g: Grid = new Map()
  for (const s of seats) {
    const k = `${Math.floor(s.x / CELL)},${Math.floor(s.y / CELL)}`
    if (!g.has(k)) g.set(k, [])
    g.get(k)!.push(s)
  }
  return g
}

function findNearest(
  grid: Grid,
  wx: number,
  wy: number,
  maxDist: number
): Seat | null {
  const cx = Math.floor(wx / CELL)
  const cy = Math.floor(wy / CELL)
  let best: Seat | null = null
  let bestD = maxDist * maxDist
  for (let dx = -1; dx <= 1; dx++) {
    for (let dy = -1; dy <= 1; dy++) {
      const bucket = grid.get(`${cx + dx},${cy + dy}`)
      if (!bucket) continue
      for (const s of bucket) {
        const d = (s.x - wx) ** 2 + (s.y - wy) ** 2
        if (d < bestD) {
          bestD = d
          best = s
        }
      }
    }
  }
  return best
}

// ─── component ────────────────────────────────────────────────────────────────
interface SeatsCanvasProps {
  stageRef: React.RefObject<Konva.Stage | null>
}

const SeatsCanvas: React.FC<SeatsCanvasProps> = ({ stageRef }) => {
  const shapeRef = useRef<Konva.Shape | null>(null)
  const hoveredRef = useRef<Seat | null>(null) // current hovered seat (no re-render)

  const { seats, toggleSeat, setTooltip, zoomToSeat, isZoomedIn } =
    useBookingStore()

  // Rebuild spatial grid whenever seats array changes (status updates included)
  const grid = useMemo(() => buildGrid(seats), [seats])

  // ── draw function — called by Konva when layer needs repaint ───────────────
  const drawScene = useCallback(
    (ctx: Konva.Context) => {
      const canvas = ctx.canvas as unknown as HTMLCanvasElement
      const raw = canvas.getContext("2d")
      if (!raw) return

      raw.clearRect(0, 0, canvas.width, canvas.height)

      const hov = hoveredRef.current

      for (const s of seats) {
        const isHov = hov?.id === s.id
        const isSel = s.status === "selected"

        // Radius
        const r = isHov ? R_HOV : isSel ? R_SEL : R

        // Fill colour
        let fill: string
        if (s.status === "taken") {
          fill = C_TAKEN
        } else if (isHov) {
          fill = C_HOV
        } else {
          fill =
            s.status === "selected"
              ? C_SEL
              : s.status === "vip"
                ? C_VIP
                : C_AVAIL
        }

        raw.globalAlpha = s.status === "taken" ? 0.22 : 1

        // Glow for selected / hovered
        if (isSel) {
          raw.shadowColor = C_SEL_GL
          raw.shadowBlur = 10
        } else if (isHov && s.status !== "taken") {
          raw.shadowColor = C_HOV
          raw.shadowBlur = 8
        } else {
          raw.shadowBlur = 0
        }

        raw.beginPath()
        raw.arc(s.x, s.y, r, 0, Math.PI * 2)
        raw.fillStyle = fill
        raw.fill()

        // Stroke ring for selected
        if (isSel) {
          raw.strokeStyle = "#4ade80"
          raw.lineWidth = 1.5
          raw.stroke()
        }
      }

      // Reset
      raw.globalAlpha = 1
      raw.shadowBlur = 0
    },
    [seats]
  )

  // Hit scene — invisible; Konva uses it for pointer events (we don't use it)
  const drawHit = useCallback((ctx: Konva.Context) => {
    // We handle events manually on the Stage, so no hit canvas needed
  }, [])

  // Trigger repaint whenever seats change
  useEffect(() => {
    shapeRef.current?.getLayer()?.batchDraw()
  }, [seats])

  // ── Stage-level mouse event handling ──────────────────────────────────────
  // We attach to the Stage container div directly so we bypass Konva's per-node
  // event dispatch (which is slow with 1000+ nodes).

  const getWorldPos = useCallback(
    (e: MouseEvent | TouchEvent): { x: number; y: number } | null => {
      const stage = stageRef.current
      if (!stage) return null
      const rect = (stage.container() as HTMLElement).getBoundingClientRect()
      const scale = stage.scaleX()
      const pos = stage.position()
      let clientX: number, clientY: number
      if ("touches" in e) {
        clientX = e.touches[0].clientX
        clientY = e.touches[0].clientY
      } else {
        clientX = e.clientX
        clientY = e.clientY
      }
      return {
        x: (clientX - rect.left - pos.x) / scale,
        y: (clientY - rect.top - pos.y) / scale,
      }
    },
    [stageRef]
  )

  const handleMouseMove = useCallback(
    (e: MouseEvent) => {
      const wp = getWorldPos(e)
      if (!wp) return

      const hit = findNearest(grid, wp.x, wp.y, R_HOV + 4)
      const prev = hoveredRef.current

      if (hit?.id !== prev?.id) {
        hoveredRef.current = hit ?? null
        shapeRef.current?.getLayer()?.batchDraw()

        const container = stageRef.current?.container() as
          | HTMLElement
          | undefined
        if (container)
          container.style.cursor =
            hit && hit.status !== "taken" ? "pointer" : "grab"
      }

      if (hit && hit.status !== "taken") {
        const rect = (
          stageRef.current!.container() as HTMLElement
        ).getBoundingClientRect()
        setTooltip({
          visible: true,
          x: rect.left + (e.clientX - rect.left) + 14,
          y: rect.top + (e.clientY - rect.top) - 14,
          seat: hit,
        })
      } else {
        setTooltip({ visible: false, x: 0, y: 0, seat: null })
      }
    },
    [grid, getWorldPos, setTooltip, stageRef]
  )

  const handleMouseLeave = useCallback(() => {
    hoveredRef.current = null
    shapeRef.current?.getLayer()?.batchDraw()
    setTooltip({ visible: false, x: 0, y: 0, seat: null })
    const container = stageRef.current?.container() as HTMLElement | undefined
    if (container) container.style.cursor = "grab"
  }, [setTooltip, stageRef])

  const handleClick = useCallback(
    (e: MouseEvent) => {
      const wp = getWorldPos(e)
      if (!wp) return
      const hit = findNearest(grid, wp.x, wp.y, R_HOV + 4)
      if (!hit || hit.status === "taken") return

      const stage = stageRef.current
      if (!stage) return

      if (!isZoomedIn) {
        zoomToSeat(hit, stage.width(), stage.height())
      } else {
        toggleSeat(hit.id)
      }
    },
    [grid, getWorldPos, isZoomedIn, zoomToSeat, toggleSeat, stageRef]
  )

  // Attach / detach native event listeners on the Stage container
  useEffect(() => {
    const el = stageRef.current?.container() as HTMLElement | undefined
    if (!el) return
    el.addEventListener("mousemove", handleMouseMove, { passive: true })
    el.addEventListener("mouseleave", handleMouseLeave)
    el.addEventListener("click", handleClick)
    return () => {
      el.removeEventListener("mousemove", handleMouseMove)
      el.removeEventListener("mouseleave", handleMouseLeave)
      el.removeEventListener("click", handleClick)
    }
  }, [stageRef, handleMouseMove, handleMouseLeave, handleClick])

  return (
    <Layer listening={false}>
      <Shape
        ref={shapeRef}
        sceneFunc={drawScene}
        hitFunc={drawHit}
        listening={false}
        perfectDrawEnabled={false}
      />
    </Layer>
  )
}

export default SeatsCanvas
