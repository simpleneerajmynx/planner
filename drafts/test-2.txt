"use client"

import React from "react"

// Deterministic pseudo-random generator so SSG/SSR safely matches Client without flashing
const Seat = ({ id }: { id: number }) => {
  const isAvailable = (id * 16807) % 2147483647 > 600000000

  return (
    <div
      className={`relative h-[7px] w-[7px] cursor-pointer rounded-full transition-all duration-150 hover:z-10 hover:scale-[1.8] ${
        isAvailable ? "bg-[#1f56d2]" : "bg-[#dededf]"
      }`}
      title={`Seat ID: ${id}`}
    />
  )
}

const SeatGrid = ({
  rows,
  cols,
  seed,
}: {
  rows: number
  cols: number
  seed: number
}) => {
  return (
    <div
      className="grid w-full justify-items-center gap-[5px]"
      style={{ gridTemplateColumns: `repeat(${cols}, minmax(0, 1fr))` }}
    >
      {Array.from({ length: rows * cols }).map((_, i) => (
        <Seat key={i} id={seed * 1000 + i + 1} />
      ))}
    </div>
  )
}

const FloorColumn = ({
  layout,
  cols,
  topRound = false,
  seedBase,
}: {
  layout: number[]
  cols: number
  topRound?: boolean
  seedBase: number
}) => (
  <div className="flex flex-col gap-2.5">
    {layout.map((rows, idx) => (
      <div
        key={idx}
        className={`bg-white p-3.5 shadow-sm ${
          idx === 0 && topRound ? "rounded-t-[30px]" : "rounded-md"
        } ${idx === layout.length - 1 ? "rounded-b-[30px]" : ""}`}
      >
        <SeatGrid rows={rows} cols={cols} seed={seedBase * 10 + idx} />
      </div>
    ))}
  </div>
)

const SidebarSection = ({
  title,
  widthClass = "w-[90px]",
  seed,
}: {
  title: string
  widthClass?: string
  seed: number
}) => (
  <div className={`flex flex-col ${widthClass}`}>
    <div className="flex h-[52px] items-center justify-center rounded-t-lg bg-[#e4e4e6] text-[22px] font-bold tracking-wide text-[#7a7a82] select-none">
      {title}
    </div>
    <div className="flex flex-1 flex-col items-center gap-3.5 rounded-b-lg bg-white p-3.5 shadow-sm">
      <SeatGrid rows={16} cols={5} seed={seed} />
      <SeatGrid rows={18} cols={5} seed={seed + 1} />
    </div>
  </div>
)

const CurvedSection = ({
  rotate,
  dy,
  dx = 0,
  cols = 6,
  seed,
}: {
  rotate: number
  dy: number
  dx?: number
  cols?: number
  seed: number
}) => (
  <div
    className="flex flex-col items-center rounded-t-[12px] rounded-b-[24px] border border-gray-50 bg-white p-3.5 shadow-sm"
    style={{
      transform: `translate(${dx}px, ${dy}px) rotate(${rotate}deg)`,
      transformOrigin: "top center",
    }}
  >
    {/* Grey header visual placeholder */}
    <div className="mb-3 h-[4px] w-[50%] rounded-full bg-[#e4e4e6]" />
    <SeatGrid rows={14} cols={cols} seed={seed} />
  </div>
)

export default function SeatingMap() {
  return (
    <div className="relative flex w-full justify-center bg-[#f4f4f7] font-sans">
      {/* Main Interactive Map Container */}
      <div className="relative flex w-full max-w-[1200px] origin-top scale-[0.8] flex-col items-center px-4 pt-20 pb-32 select-none sm:scale-[0.9] lg:scale-100">
        {/* Main Columns Wrapper */}
        <div className="flex items-start justify-center gap-3">
          {/* Left Wings */}
          <div className="mt-[110px] flex gap-2.5">
            <SidebarSection title="315" seed={315} />
            <SidebarSection title="216" seed={216} />
          </div>

          {/* Center Floor Architecture */}
          <div className="relative z-10 mx-2 flex flex-col items-center">
            {/* STAGE Block */}
            <div className="mb-6 flex h-[85px] w-[360px] items-center justify-center rounded-[3px] bg-[#222222] text-[34px] font-extrabold tracking-[0.08em] text-white shadow-md">
              STAGE
            </div>

            {/* Structured Seat Grids */}
            <div className="flex gap-2.5">
              <FloorColumn layout={[8, 12, 10]} cols={8} seedBase={1} />
              <FloorColumn
                layout={[10, 12, 10]}
                cols={14}
                topRound
                seedBase={2}
              />
              <FloorColumn layout={[8, 12, 10]} cols={8} seedBase={3} />
            </div>
          </div>

          {/* Right Wings */}
          <div className="mt-[110px] flex gap-2.5">
            <SidebarSection title="201" seed={201} />
            <SidebarSection title="301" seed={301} />
          </div>
        </div>

        {/* Lower Arena Bowl / Curved Section */}
        <div className="relative mt-5 flex h-[400px] w-full items-start justify-center gap-1.5">
          <CurvedSection rotate={-38} dy={-65} dx={45} seed={101} />
          <CurvedSection rotate={-22} dy={-25} dx={22} seed={102} />
          <CurvedSection rotate={-9} dy={-5} dx={10} seed={103} />
          {/* Center Curve with extra width */}
          <CurvedSection rotate={0} dy={5} dx={0} cols={8} seed={104} />
          <CurvedSection rotate={9} dy={-5} dx={-10} seed={105} />
          <CurvedSection rotate={22} dy={-25} dx={-22} seed={106} />
          <CurvedSection rotate={38} dy={-65} dx={-45} seed={107} />
        </div>
      </div>
    </div>
  )
}
