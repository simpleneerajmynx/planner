"use client"

import React from "react"
import { useBookingStore } from "@/store/bookingStore"

interface ZoomControlsProps {
  canvasWidth: number
  canvasHeight: number
}

const ZoomControls: React.FC<ZoomControlsProps> = ({
  canvasWidth,
  canvasHeight,
}) => {
  const { stageScale, stagePos, setStageTransform, resetZoom } =
    useBookingStore()

  const zoomIn = () => {
    const newScale = Math.min(8, stageScale * 1.25)
    const centerX = canvasWidth / 2
    const centerY = canvasHeight / 2
    const newX = centerX - (centerX - stagePos.x) * (newScale / stageScale)
    const newY = centerY - (centerY - stagePos.y) * (newScale / stageScale)
    setStageTransform(newScale, { x: newX, y: newY })
  }

  const zoomOut = () => {
    const newScale = Math.max(0.5, stageScale / 1.25)
    const centerX = canvasWidth / 2
    const centerY = canvasHeight / 2
    const newX = centerX - (centerX - stagePos.x) * (newScale / stageScale)
    const newY = centerY - (centerY - stagePos.y) * (newScale / stageScale)
    setStageTransform(newScale, { x: newX, y: newY })
  }

  const btnStyle: React.CSSProperties = {
    width: 32,
    height: 32,
    display: "flex",
    alignItems: "center",
    justifyContent: "center",
    background: "var(--bg-elevated)",
    border: "1px solid var(--border-medium)",
    borderRadius: 8,
    color: "var(--text-secondary)",
    cursor: "pointer",
    transition: "all 0.15s ease",
    fontFamily: "var(--font-mono)",
  }

  return (
    <div
      className="absolute top-4 right-4 flex flex-col gap-1.5"
      style={{ zIndex: 10 }}
    >
      <button style={btnStyle} onClick={zoomIn} title="Zoom in">
        <svg
          width="14"
          height="14"
          viewBox="0 0 24 24"
          fill="none"
          stroke="currentColor"
          strokeWidth="2.5"
        >
          <circle cx="11" cy="11" r="8" />
          <path d="M21 21l-4.35-4.35M11 8v6M8 11h6" />
        </svg>
      </button>
      <button style={btnStyle} onClick={zoomOut} title="Zoom out">
        <svg
          width="14"
          height="14"
          viewBox="0 0 24 24"
          fill="none"
          stroke="currentColor"
          strokeWidth="2.5"
        >
          <circle cx="11" cy="11" r="8" />
          <path d="M21 21l-4.35-4.35M8 11h6" />
        </svg>
      </button>
      <button style={btnStyle} onClick={resetZoom} title="Reset view">
        <svg
          width="14"
          height="14"
          viewBox="0 0 24 24"
          fill="none"
          stroke="currentColor"
          strokeWidth="2"
        >
          <path d="M3 12a9 9 0 1 0 18 0 9 9 0 0 0-18 0" />
          <path d="M9 12l3-3 3 3" />
          <path d="M12 15V9" />
        </svg>
      </button>
    </div>
  )
}

export default ZoomControls
