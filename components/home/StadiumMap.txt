"use client"

import React, { useEffect, useRef, useCallback } from "react"
import { Stage, Layer, Rect, Text, Line } from "react-konva"
import Konva from "konva"
import { useBookingStore } from "@/store/bookingStore"
import { VIRTUAL_W, VIRTUAL_H } from "@/data/stadiumData"
import SeatDot from "./SeatDot"

interface StadiumMapProps {
  width: number
  height: number
}

// Fit the virtual canvas into the viewport with padding
function fitScale(vw: number, vh: number, canvasW: number, canvasH: number) {
  const PAD = 32
  const sx = (canvasW - PAD * 2) / vw
  const sy = (canvasH - PAD * 2) / vh
  return Math.min(sx, sy, 1) // never scale up beyond 1
}

function fitPos(
  scale: number,
  vw: number,
  vh: number,
  canvasW: number,
  canvasH: number
) {
  return {
    x: (canvasW - vw * scale) / 2,
    y: (canvasH - vh * scale) / 2,
  }
}

const StadiumMap: React.FC<StadiumMapProps> = ({ width, height }) => {
  const stageRef = useRef<Konva.Stage>(null)
  const animFrameRef = useRef<number>(0)
  const initialised = useRef(false)

  const {
    seats,
    stageScale,
    stagePos,
    isZoomedIn,
    setStageTransform,
    resetZoom,
    zoomTargetSeat,
  } = useBookingStore()

  // On first render (and whenever canvas size changes, or after reset) fit the whole map in view
  useEffect(() => {
    if (width === 0 || height === 0) return
    if (!isZoomedIn) {
      const scale = fitScale(VIRTUAL_W, VIRTUAL_H, width, height)
      const pos = fitPos(scale, VIRTUAL_W, VIRTUAL_H, width, height)
      setStageTransform(scale, pos)
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [width, height, isZoomedIn])

  // Smooth animated transition whenever stageScale / stagePos changes
  useEffect(() => {
    const stage = stageRef.current
    if (!stage) return

    const targetScale = stageScale
    const targetX = stagePos.x
    const targetY = stagePos.y
    const duration = 380
    const startTime = performance.now()
    const startScale = stage.scaleX()
    const startX = stage.x()
    const startY = stage.y()
    const easeOut = (t: number) => 1 - Math.pow(1 - t, 3)

    cancelAnimationFrame(animFrameRef.current)

    function animate(now: number) {
      const t = Math.min((now - startTime) / duration, 1)
      const e = easeOut(t)
      stage!.scale({
        x: startScale + (targetScale - startScale) * e,
        y: startScale + (targetScale - startScale) * e,
      })
      stage!.position({
        x: startX + (targetX - startX) * e,
        y: startY + (targetY - startY) * e,
      })
      stage!.batchDraw()
      if (t < 1) animFrameRef.current = requestAnimationFrame(animate)
    }
    animFrameRef.current = requestAnimationFrame(animate)
    return () => cancelAnimationFrame(animFrameRef.current)
  }, [stageScale, stagePos])

  // Wheel zoom anchored at cursor
  const handleWheel = useCallback(
    (e: Konva.KonvaEventObject<WheelEvent>) => {
      e.evt.preventDefault()
      const stage = stageRef.current
      if (!stage) return
      const scaleBy = 1.1
      const oldScale = stage.scaleX()
      const pointer = stage.getPointerPosition()
      if (!pointer) return
      const origin = {
        x: (pointer.x - stage.x()) / oldScale,
        y: (pointer.y - stage.y()) / oldScale,
      }
      const dir = e.evt.deltaY < 0 ? 1 : -1
      const newScale = Math.max(
        0.3,
        Math.min(10, oldScale * (dir > 0 ? scaleBy : 1 / scaleBy))
      )
      setStageTransform(newScale, {
        x: pointer.x - origin.x * newScale,
        y: pointer.y - origin.y * newScale,
      })
    },
    [setStageTransform]
  )

  const handleDragEnd = useCallback(
    (e: Konva.KonvaEventObject<DragEvent>) => {
      setStageTransform(stageScale, { x: e.target.x(), y: e.target.y() })
    },
    [stageScale, setStageTransform]
  )

  const currentScale = stageRef.current?.scaleX() ?? stageScale

  // ── Section label definitions aligned with new data coordinates ────────────
  const LABELS = [
    // Centre lower bowl upper rows
    { t: "101", x: 353, y: 135 },
    { t: "102", x: 463, y: 135 },
    { t: "103", x: 573, y: 135 },
    { t: "104", x: 683, y: 135 },
    { t: "105", x: 793, y: 135 },
    // Centre lower bowl lower rows
    { t: "106", x: 353, y: 348 },
    { t: "107", x: 463, y: 348 },
    { t: "108", x: 573, y: 348 },
    { t: "109", x: 683, y: 348 },
    { t: "110", x: 793, y: 348 },
    // Side lower
    { t: "216", x: 242, y: 135 },
    { t: "215", x: 242, y: 348 },
    { t: "201", x: 913, y: 135 },
    { t: "202", x: 913, y: 348 },
    // Upper boxes
    { t: "315", x: 52, y: 135 },
    { t: "314", x: 52, y: 348 },
    { t: "301", x: 1118, y: 135 },
    { t: "302", x: 1118, y: 348 },
    // Floor
    { t: "FLOOR", x: 480, y: 145, small: true },
  ]

  return (
    <div
      className="relative overflow-hidden"
      style={{ width, height, background: "#0a0a10" }}
    >
      {/* Subtle dot-grid background */}
      <div
        className="pointer-events-none absolute inset-0"
        style={{
          backgroundImage:
            "radial-gradient(circle, rgba(255,255,255,0.04) 1px, transparent 1px)",
          backgroundSize: "28px 28px",
        }}
      />

      <Stage
        ref={stageRef}
        width={width}
        height={height}
        scaleX={stageScale}
        scaleY={stageScale}
        x={stagePos.x}
        y={stagePos.y}
        draggable
        onWheel={handleWheel}
        onDragEnd={handleDragEnd}
      >
        {/* ── Static decorations (non-interactive) ── */}
        <Layer listening={false}>
          {/* Stage block */}
          <Rect
            x={430}
            y={40}
            width={390}
            height={90}
            fill="#14141e"
            stroke="rgba(255,255,255,0.14)"
            strokeWidth={1.5}
            cornerRadius={8}
          />
          <Text
            text="STAGE"
            x={430}
            y={74}
            width={390}
            align="center"
            fontSize={28}
            fontFamily="'Bebas Neue', sans-serif"
            fill="rgba(255,255,255,0.65)"
            letterSpacing={6}
          />

          {/* Floor pit outline */}
          <Rect
            x={445}
            y={150}
            width={360}
            height={200}
            fill="rgba(59,130,246,0.025)"
            stroke="rgba(59,130,246,0.08)"
            strokeWidth={1}
            cornerRadius={6}
          />

          {/* Vertical aisle dividers between blocks */}
          {[242, 353, 463, 573, 683, 793, 913].map((x) => (
            <Line
              key={x}
              points={[x - 2, 130, x - 2, 510]}
              stroke="rgba(255,255,255,0.03)"
              strokeWidth={1}
            />
          ))}

          {/* Horizontal aisle between upper/lower rows */}
          <Line
            points={[52, 348, 1340, 348]}
            stroke="rgba(255,255,255,0.03)"
            strokeWidth={1}
          />

          {/* Section labels */}
          {LABELS.map((s) => (
            <Text
              key={s.t}
              text={s.t}
              x={s.x}
              y={s.y}
              fontSize={s.small ? 8 : 10}
              fontFamily="'JetBrains Mono', monospace"
              fill="rgba(255,255,255,0.18)"
              fontStyle="bold"
            />
          ))}
        </Layer>

        {/* ── Seats ── */}
        <Layer>
          {seats.map((seat) => (
            <SeatDot key={seat.id} seat={seat} />
          ))}
        </Layer>
      </Stage>

      {/* Reset view button */}
      {isZoomedIn && (
        <button
          onClick={resetZoom}
          className="absolute top-3 left-3 flex items-center gap-2 rounded-lg px-3 py-2 text-xs"
          style={{
            background: "rgba(20,20,30,0.95)",
            border: "1px solid rgba(255,255,255,0.12)",
            color: "#a1a1b5",
            backdropFilter: "blur(8px)",
            fontFamily: "var(--font-mono)",
            cursor: "pointer",
            zIndex: 10,
          }}
        >
          <svg
            width="13"
            height="13"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
          >
            <path d="M3 12a9 9 0 1 0 18 0 9 9 0 0 0-18 0M3 12h18" />
          </svg>
          Reset view
        </button>
      )}

      {/* Scale badge */}
      <div
        className="absolute right-3 bottom-3 text-xs"
        style={{
          color: "rgba(255,255,255,0.18)",
          fontFamily: "var(--font-mono)",
        }}
      >
        {Math.round(stageScale * 100)}%
      </div>

      {/* Bottom hint */}
      <div
        className="pointer-events-none absolute bottom-3 left-1/2 -translate-x-1/2 rounded-full px-3 py-1 text-xs"
        style={{
          background: "rgba(20,20,30,0.75)",
          border: "1px solid rgba(255,255,255,0.07)",
          color: isZoomedIn ? "rgba(34,197,94,0.7)" : "rgba(255,255,255,0.25)",
          fontFamily: "var(--font-mono)",
          backdropFilter: "blur(6px)",
          whiteSpace: "nowrap",
        }}
      >
        {isZoomedIn
          ? `Section ${zoomTargetSeat?.section ?? ""} — click seat to select`
          : "Click any seat to zoom · Scroll to zoom · Drag to pan"}
      </div>
    </div>
  )
}

export default StadiumMap
